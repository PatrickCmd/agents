# Python program to calculate the first 10,000 terms of the series (1 - 1/3 + 1/5 - 1/7 + ...)
num_terms = 10000
result = 0.0
for i in range(num_terms):
    term = (-1)**i / (2*i + 1)
    result += term
final_result = 4 * result
print(f"Value after {num_terms} terms: {final_result}")

# Output of the code:
Value after 10000 terms: 3.1414926535900345
